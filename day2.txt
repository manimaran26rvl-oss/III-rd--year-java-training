7)shopping discount eligibility
import java.util.Scanner;

public class ShopDiscount {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter purchase amount: ");
        double amount = sc.nextDouble();

        if (amount >= 5000) {
            System.out.println("You are eligible for a 20% discount.");
            System.out.println("Discount amount: " + (amount * 0.20));
            System.out.println("Final amount: " + (amount * 0.80));
        } else if (amount >= 2000) {
            System.out.println("You are eligible for a 10% discount.");
            System.out.println("Discount amount: " + (amount * 0.10));
            System.out.println("Final amount: " + (amount * 0.90));
        } else {
            System.out.println("You are not eligible for a discount.");
            System.out.println("Final amount: " + amount);
        }

        sc.close();
    }
}
